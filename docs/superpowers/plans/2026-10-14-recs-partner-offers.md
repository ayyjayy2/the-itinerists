# Partner offers on Tips & Recs: implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Two soft cards on Tips & Recs, "Need data abroad?" (eSIMs) and "Covered if plans change?"
(insurance), for people whose trip leaves their phone's country. Each opens a pop-up with partner
links and codes, plus a plain-words "What's an eSIM?" tab. Taps are counted per partner for Pulse.

**Architecture:** The offers come from `_config/partners` (Firestore, written by an admin script from
a gitignored file), read once by `PartnersService`. A pure `abroad.ts` decides who sees the cards,
from each stop's time zone → country (a generated table) against the phone's zone. A standalone
`PartnerOffersComponent` renders the cards and pop-ups at the top of Recs. Taps go to `_activity` as
`type: 'offer'`.

**Tech stack:** Angular 20 (standalone, signals), Firebase JS SDK 12, Karma/Jasmine, node:test
(rules, Pulse, scripts), Playwright (demo build).

Spec: `docs/superpowers/specs/2026-10-14-recs-partner-offers-design.md` · Issue: #474

```ts
import { ZONE_COUNTRIES } from './zone-countries';

/**
 * Is a trip abroad for this person? Each stop's time zone gives its country,
 * compared with the country of the phone's own zone (Tips & Recs partner
 * offers, #474). US territories count as the US: a US phone going to San Juan
 * isn't abroad.
 */
const US_TERRITORIES = new Set(['PR', 'VI', 'GU', 'AS', 'MP', 'UM']);

function countryOf(zone: string | null | undefined): string | null {
  const code = zone ? ZONE_COUNTRIES[zone] : undefined;
  if (!code) return null;
  return US_TERRITORIES.has(code) ? 'US' : code;
}

/** The stops' countries that differ from the phone's, in stop order, once each. */
export function foreignCountries(stopZones: ReadonlyArray<string | null | undefined>, deviceZone: string): string[] {
  const home = countryOf(deviceZone);
  if (!home) return [];
  const out: string[] = [];
  for (const z of stopZones) {
    const c = countryOf(z);
    if (c && c !== home && !out.includes(c)) out.push(c);
  }
  return out;
}

let names: Intl.DisplayNames | null | undefined;
/** "DE" → "Germany"; the code itself when the browser can't name it. */
export function countryName(code: string): string {
  if (names === undefined) {
    try { names = new Intl.DisplayNames(['en'], { type: 'region' }); } catch { names = null; }
  }
  const n = names?.of(code);
  return n && n !== code ? n : code;
}

/** "Germany", "Germany & Czechia", "Germany, Czechia + 2 more". */
export function countryList(list: readonly string[]): string {
  if (list.length <= 1) return list[0] ?? '';
  if (list.length === 2) return `${list[0]} & ${list[1]}`;
  return `${list[0]}, ${list[1]} + ${list.length - 2} more`;
}
```

```ts
import { Injectable, InjectionToken, Injector, inject, runInInjectionContext, signal } from '@angular/core';
import { Firestore, doc, getDoc } from '@angular/fire/firestore';
import { PartnerOffer, PartnersConfig } from '../models/trip.models';
import { safeLink } from '../utils/safe-link';
import { deviceZone } from '../utils/zones';

/** The phone's own time zone (a token so tests can choose it). */
export const DEVICE_ZONE = new InjectionToken<string>('DEVICE_ZONE', { factory: deviceZone });

const ID = /^[a-z0-9-]{1,32}$/;
function cleanOffer(p: Partial<PartnerOffer> | null | undefined): PartnerOffer | null {
  if (!p || !ID.test(p.id ?? '') || !p.name || !p.offer) return null;
  const url = safeLink(p.url);
  if (!url || !url.startsWith('https://')) return null;
  return { id: p.id!, name: p.name, offer: p.offer, url, ...(p.code ? { code: p.code } : {}) };
}

/** Only well-formed offers survive; switched off or missing means none. */
export function cleanPartners(raw: Partial<PartnersConfig> | null | undefined): PartnersConfig | null {
  if (!raw?.active) return null;
  const list = (l: unknown) => (Array.isArray(l) ? l : []).map(cleanOffer).filter((p): p is PartnerOffer => !!p);
  return { active: true, esim: list(raw.esim), insurance: list(raw.insurance) };
}

/**
 * Partner offers for trips abroad (Tips & Recs, #474), from `_config/partners`
 * (scripts/set-partners.js). Read once per session: offers rarely change.
 */
@Injectable({ providedIn: 'root' })
export class PartnersService {
  private firestore = inject(Firestore);
  private injector  = inject(Injector);
  private readonly _config = signal<PartnersConfig | null>(null);
  readonly config = this._config.asReadonly();
  private loading?: Promise<void>;

  ensureLoaded(): Promise<void> {
    return this.loading ??= this.load()
      .then(raw => this._config.set(cleanPartners(raw)))
      .catch(err => console.warn('[Partners] offers unavailable:', (err as Error).message));
  }

  private async load(): Promise<Partial<PartnersConfig> | null> {
    const snap = await runInInjectionContext(this.injector, () => getDoc(doc(this.firestore, '_config', 'partners')));
    return snap.exists() ? snap.data() as Partial<PartnersConfig> : null;
  }
}
```

type CardId = 'esim' | 'insurance';
interface OfferCard { id: CardId; icon: string; title: string; subline: string; partners: PartnerOffer[]; }

const HIDDEN_KEY = (tripId: string) => `itinerists_offers_hidden_${tripId}`;

/**
 * Partner offers for trips abroad, at the top of Tips & Recs (#474): a soft card
 * for eSIMs and one for travel insurance, each opening a pop-up. Shown only when
 * a stop is in another country from this phone; each card can be hidden per trip.

@Component({
  selector: 'app-partner-offers',
  standalone: true,
  templateUrl: './partner-offers.component.html',
  styleUrl: './partner-offers.component.scss',
})
export class PartnerOffersComponent {
  private partners = inject(PartnersService);
  private trips    = inject(TripService);
  private usage    = inject(UsageService);
  private zone     = inject(DEVICE_ZONE);

  readonly guideIntro = ESIM_INTRO;
  readonly guideSteps = ESIM_STEPS;
  readonly guideNotes = ESIM_GOOD_TO_KNOW;

  private readonly tripId = computed(() => this.trips.activeTrip()?.id ?? null);
  private readonly hidden = signal<Set<CardId>>(new Set());
  readonly openCard = signal<CardId | null>(null);
  readonly tab      = signal<'pick' | 'guide'>('pick');
  readonly copied   = signal<string | null>(null);
  private trigger: HTMLElement | null = null;
  private readonly closeBtn = viewChild<ElementRef<HTMLButtonElement>>('closeBtn');

  /** The trip's countries outside this phone's, as names. */
  readonly abroad = computed(() => {
    const t = this.trips.activeTrip();
    if (!t) return [];
    const zones = [...tripDestinations(t).map(d => d.timeZone), t.timeZone];
    return foreignCountries(zones, this.zone).map(countryName);
  });

  readonly cards = computed((): OfferCard[] => {
    const cfg = this.partners.config();
    const places = this.abroad();
    if (!cfg || !places.length) return [];
    const all: OfferCard[] = [
      { id: 'esim', icon: '📶', title: 'Need data abroad?', subline: `Skip roaming fees in ${countryList(places)}`, partners: cfg.esim },
      { id: 'insurance', icon: '🛡', title: 'Covered if plans change?', subline: 'Travel insurance for trips abroad', partners: cfg.insurance },
    ];
    return all.filter(c => c.partners.length && !this.hidden().has(c.id));
  });

  readonly open = computed(() => this.cards().find(c => c.id === this.openCard()) ?? null);

  constructor() {
    void this.partners.ensureLoaded();
    effect(() => {                                   // hidden cards, per trip, on this device
      const id = this.tripId();
      let list: CardId[] = [];
      try { list = id ? JSON.parse(localStorage.getItem(HIDDEN_KEY(id)) ?? '[]') : []; } catch { /* none */ }
      this.hidden.set(new Set(list));
    });
    effect(() => { if (this.open()) this.closeBtn()?.nativeElement.focus(); });
  }

  show(card: OfferCard, ev: Event): void {
    this.trigger = ev.currentTarget as HTMLElement;
    this.tab.set('pick');
    this.copied.set(null);
    this.openCard.set(card.id);
  }

  close(): void {
    this.openCard.set(null);
    this.trigger?.focus();
  }

  hide(card: OfferCard, ev: Event): void {
    ev.stopPropagation();
    const next = new Set(this.hidden()).add(card.id);
    this.hidden.set(next);
    const id = this.tripId();
    try { if (id) localStorage.setItem(HIDDEN_KEY(id), JSON.stringify([...next])); } catch { /* storage unavailable */ }
  }

  async copy(code: string, el: HTMLElement): Promise<void> {
    try {
      await navigator.clipboard.writeText(code);
    } catch {
      const range = document.createRange();             // can't copy: select it, so a long-press copies
      range.selectNodeContents(el);
      const sel = window.getSelection(); sel?.removeAllRanges(); sel?.addRange(range);
      return;
    }
    this.copied.set(code);
    setTimeout(() => { if (this.copied() === code) this.copied.set(null); }, 2000);
  }

  opened(p: PartnerOffer): void { this.usage.recordOffer(p.id); }

  @HostListener('document:keydown.escape')
  onEscape(): void { if (this.openCard()) this.close(); }
}

/** Partner-offer taps on Tips & Recs (#474): taps and people per partner, most tapped first. */
export function offerStats(rows) {
  const by = new Map();
  for (const r of rows) {
    if (r.type !== 'offer' || !r.offer) continue;
    const s = by.get(r.offer) ?? by.set(r.offer, { taps: 0, users: new Set() }).get(r.offer);
    s.taps++; s.users.add(r.uid);
  }
  return [...by.entries()].map(([offer, s]) => ({ offer, taps: s.taps, users: s.users.size }))
    .sort((a, b) => b.taps - a.taps || a.offer.localeCompare(b.offer));
}
